function classifyPixel(r, g, b) {
  const rn = r / 255, gn = g / 255, bn = b / 255;
  const max = Math.max(rn, gn, bn), min = Math.min(rn, gn, bn);
  const d = max - min;
  const v = max;
  const s = max === 0 ? 0 : d / max;

  if (v < 0.20) return "Black";
  if (s < 0.16 && v > 0.72) return "White";
  if (s < 0.12) return v > 0.5 ? "White" : "Black";

  let h = 0;
  if (d !== 0) {
    if (max === rn) h = ((gn - bn) / d) % 6;
    else if (max === gn) h = (bn - rn) / d + 2;
    else h = (rn - gn) / d + 4;
    h = Math.round(h * 60);
    if (h < 0) h += 360;
  }

  if (h < 16 || h >= 335) return "Red";
  if (h < 40) return "Orange";
  if (h < 72) return "Yellow";
  if (h < 168) return "Green";
  if (h < 265) return "Blue";
  return "Red";
}

// Build interactive coloring-book state whenever the selected country changes
flagState = {
  const img = new Image();
  img.src = selected.path;
  await img.decode();

  const w = 480;
  const h = Math.round((img.height / img.width) * w);
  const srcCanvas = document.createElement("canvas");
  srcCanvas.width = w;
  srcCanvas.height = h;
  const srcCtx = srcCanvas.getContext("2d");
  srcCtx.drawImage(img, 0, 0, w, h);
  const srcData = srcCtx.getImageData(0, 0, w, h).data;

  const n = w * h;
  const trueClass = new Array(n);
  const userClass = new Array(n).fill(null);
  const isEdge = new Uint8Array(n);

  for (let i = 0; i < n; i++) {
    const idx = i * 4;
    if (srcData[idx + 3] < 128) {
      trueClass[i] = null;
    } else {
      trueClass[i] = classifyPixel(srcData[idx], srcData[idx + 1], srcData[idx + 2]);
    }
  }

  // Detect shape boundaries so stripes, stars, and emblems have crisp dark outlines
  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      const i = y * w + x;
      const c = trueClass[i];
      if (!c) continue;
      let edge = false;
      for (let dy = -1; dy <= 1 && !edge; dy++) {
        for (let dx = -1; dx <= 1; dx++) {
          const nx = x + dx, ny = y + dy;
          if (nx >= 0 && nx < w && ny >= 0 && ny < h) {
            const ni = ny * w + nx;
            if (trueClass[ni] !== c) {
              edge = true;
              break;
            }
          }
        }
      }
      if (edge) isEdge[i] = 1;
    }
  }

  return { w, h, trueClass, userClass, isEdge, srcData };
}

mutable paintVersion = 0

// Draw the current coloring-book canvas and handle clicks to fill shapes
coloringBoard = {
  const { w, h, trueClass, userClass, isEdge } = flagState;
  const canvas = document.createElement("canvas");
  canvas.width = w;
  canvas.height = h;
  canvas.style.maxWidth = "480px";
  canvas.style.width = "100%";
  canvas.style.height = "auto";
  canvas.style.cursor = "crosshair";
  canvas.style.border = "2px solid #475569";
  canvas.style.borderRadius = "6px";
  canvas.style.boxShadow = "0 4px 12px rgba(0,0,0,0.1)";

  const ctx = canvas.getContext("2d");
  const colorMap = Object.fromEntries(colorPalette.map(c => [c.name, c.rgb]));

  function renderCanvas() {
    const out = ctx.createImageData(w, h);
    const d = out.data;
    for (let y = 0; y < h; y++) {
      for (let x = 0; x < w; x++) {
        const i = y * w + x;
        const idx = i * 4;
        if (!trueClass[i]) {
          d[idx + 3] = 0;
          continue;
        }
        if (isEdge[i]) {
          // Dark outline for shapes inside the flag
          d[idx] = 45;
          d[idx + 1] = 55;
          d[idx + 2] = 72;
          d[idx + 3] = 255;
        } else if (userClass[i]) {
          const rgb = colorMap[userClass[i]];
          d[idx] = rgb[0];
          d[idx + 1] = rgb[1];
          d[idx + 2] = rgb[2];
          d[idx + 3] = 255;
        } else {
          // Subtle checkerboard light gray so unpainted regions are distinct from White paint
          const checker = ((Math.floor(x / 8) + Math.floor(y / 8)) % 2 === 0) ? 236 : 226;
          d[idx] = checker;
          d[idx + 1] = checker + 2;
          d[idx + 2] = checker + 5;
          d[idx + 3] = 255;
        }
      }
    }
    ctx.putImageData(out, 0, 0);
  }

  canvas.addEventListener("click", (e) => {
    const rect = canvas.getBoundingClientRect();
    const clickX = Math.floor(((e.clientX - rect.left) / rect.width) * w);
    const clickY = Math.floor(((e.clientY - rect.top) / rect.height) * h);
    if (clickX < 0 || clickX >= w || clickY < 0 || clickY >= h) return;

    const startIdx = clickY * w + clickX;
    const targetTrue = trueClass[startIdx];
    if (!targetTrue) return;

    const chosenColor = activeColor.name;

    // Flood-fill the connected shape that was clicked
    const visited = new Uint8Array(w * h);
    const queue = new Int32Array(w * h);
    let head = 0, tail = 0;
    queue[tail++] = startIdx;
    visited[startIdx] = 1;

    while (head < tail) {
      const curr = queue[head++];
      userClass[curr] = chosenColor;
      const cx = curr % w;
      const cy = (curr - cx) / w;

      if (cx > 0) {
        const ni = curr - 1;
        if (!visited[ni] && trueClass[ni] === targetTrue) { visited[ni] = 1; queue[tail++] = ni; }
      }
      if (cx < w - 1) {
        const ni = curr + 1;
        if (!visited[ni] && trueClass[ni] === targetTrue) { visited[ni] = 1; queue[tail++] = ni; }
      }
      if (cy > 0) {
        const ni = curr - w;
        if (!visited[ni] && trueClass[ni] === targetTrue) { visited[ni] = 1; queue[tail++] = ni; }
      }
      if (cy < h - 1) {
        const ni = curr + w;
        if (!visited[ni] && trueClass[ni] === targetTrue) { visited[ni] = 1; queue[tail++] = ni; }
      }
    }

    renderCanvas();
    mutable paintVersion = mutable paintVersion + 1;
  });

  renderCanvas();
  return canvas;
}

scoreStats = {
  paintVersion;
  const { trueClass, userClass, isEdge } = flagState;
  let total = 0, painted = 0, correct = 0;
  for (let i = 0; i < trueClass.length; i++) {
    if (!trueClass[i] || isEdge[i]) continue;
    total++;
    if (userClass[i]) {
      painted++;
      if (userClass[i] === trueClass[i]) correct++;
    }
  }
  return {
    pctPainted: total ? Math.round((painted / total) * 100) : 0,
    pctCorrect: total ? Math.round((correct / total) * 100) : 0
  };
}
