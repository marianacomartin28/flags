// Classify an RGB pixel into one of the 7 main flag colors
function classifyPixel(r, g, b) {
  const rn = r / 255, gn = g / 255, bn = b / 255;
  const max = Math.max(rn, gn, bn), min = Math.min(rn, gn, bn);
  const d = max - min;
  const v = max;
  const s = max === 0 ? 0 : d / max;

  if (v < 0.18) return "Black";
  if (s < 0.15 && v > 0.75) return "White";
  if (s < 0.15) return null;

  let h = 0;
  if (d !== 0) {
    if (max === rn) h = ((gn - bn) / d) % 6;
    else if (max === gn) h = (bn - rn) / d + 2;
    else h = (rn - gn) / d + 4;
    h = Math.round(h * 60);
    if (h < 0) h += 360;
  }

  if (h < 16 || h >= 335) return "Red";
  if (h < 40) return "Orange";
  if (h < 72) return "Yellow";
  if (h < 168) return "Green";
  if (h < 265) return "Blue";
  return "Red";
}

// Analyze the selected flag image and build an empty outline canvas
flagAnalysis = {
  const img = new Image();
  img.src = selected.path;
  await img.decode();

  const w = 360;
  const h = Math.round((img.height / img.width) * w);
  const canvas = document.createElement("canvas");
  canvas.width = w;
  canvas.height = h;
  const ctx = canvas.getContext("2d");
  ctx.drawImage(img, 0, 0, w, h);

  const imgData = ctx.getImageData(0, 0, w, h);
  const data = imgData.data;
  const counts = {};
  let total = 0;

  for (let i = 0; i < data.length; i += 4) {
    if (data[i + 3] < 128) continue;
    total++;
    const c = classifyPixel(data[i], data[i + 1], data[i + 2]);
    if (c) counts[c] = (counts[c] || 0) + 1;
  }

  const actualColors = Object.entries(counts)
    .filter(([_, count]) => count / total >= 0.01)
    .sort((a, b) => b[1] - a[1])
    .map(([name]) => name);

  // Create an empty white flag with light gray outlines where colors change
  const outCanvas = document.createElement("canvas");
  outCanvas.width = w;
  outCanvas.height = h;
  const outCtx = outCanvas.getContext("2d");
  const outData = outCtx.createImageData(w, h);

  for (let y = 0; y < h; y++) {
    for (let x = 0; x < w; x++) {
      const idx = (y * w + x) * 4;
      let isEdge = false;
      if (x < w - 1 && y < h - 1) {
        const idxR = (y * w + (x + 1)) * 4;
        const idxD = ((y + 1) * w + x) * 4;
        const diffR = Math.abs(data[idx] - data[idxR]) + Math.abs(data[idx + 1] - data[idxR + 1]) + Math.abs(data[idx + 2] - data[idxR + 2]);
        const diffD = Math.abs(data[idx] - data[idxD]) + Math.abs(data[idx + 1] - data[idxD + 1]) + Math.abs(data[idx + 2] - data[idxD + 2]);
        if (diffR > 45 || diffD > 45) isEdge = true;
      }
      const val = isEdge ? 180 : 255;
      outData.data[idx] = val;
      outData.data[idx + 1] = val;
      outData.data[idx + 2] = val;
      outData.data[idx + 3] = data[idx + 3];
    }
  }
  outCtx.putImageData(outData, 0, 0);

  return {
    actualColors,
    outlineUrl: outCanvas.toDataURL()
  };
}
